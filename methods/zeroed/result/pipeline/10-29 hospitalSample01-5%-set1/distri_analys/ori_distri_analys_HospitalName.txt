        if pd.isna(val):
            return None
        sv = str(val).strip()
        if sv == '':
            return None
        # If ends with percent, drop the %
        if sv.endswith('%'):
            sv = sv[:-1].strip()
        # Remove commas for thousands, etc.
        sv = sv.replace(',', '')
        # Direct parse
        try:
            return float(sv)
        except ValueError:
            # Fallback: extract first numeric token present
            m = re.search(r'-?\d+(?:\.\d+)?', sv)
            if m:
                try:
                    return float(m.group())
                except ValueError:
                    return None
            return None

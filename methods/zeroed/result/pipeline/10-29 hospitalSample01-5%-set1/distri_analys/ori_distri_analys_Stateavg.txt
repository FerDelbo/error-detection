def distr_analysis_pattern_match(dirty_csv, attr_name):
    if attr_name not in dirty_csv.columns:
        return f"Attribute '{attr_name}' not found in dataframe columns: {list(dirty_csv.columns)}"
    
    series = dirty_csv[attr_name]
    values = []
    for v in series:
        if pd.isna(v) or (isinstance(v, str) and v.strip() == ''):
            values.append('')
        else:
            values.append(str(v).strip())
    
    total_non_empty = len([v for v in values if v != ''])
    non_empty_vals = [v for v in values if v != '']
    
    patterns = [
        ('two_letters_code', re.compile(r'^[A-Za-z]{2}_[A-Za-z0-9\-]+$')),
        ('digits_only',       re.compile(r'^\d+$')),
        ('percent',           re.compile(r'^\d+%$')),
        ('alpha_space',         re.compile(r'^[A-Za-z]+(?: [A-Za-z]+)*$')),
    ]
    
    counts = Counter()
    by_pattern = {}
    for v in non_empty_vals:
        matched = None
        for name, regex in patterns:
            if regex.match(v):
                matched = name
                break
        if matched is None:
            matched = 'unmatched'
        counts[matched] += 1
        by_pattern.setdefault(matched, []).append(v)
    
    top_patterns = counts.most_common(3)
    lines = [
        f"Attribute: {attr_name}",
        f"Total non-empty values: {total_non_empty}",
        "Pattern distribution (top patterns):"
    ]
    for name, c in top_patterns:
        samples = by_pattern.get(name, [])[:3]
        lines.append(f"  - {name}: {c} occurrences; examples: {samples}")
    
    unmatched = counts.get('unmatched', 0)
    lines.append(f"Unmatched (no pattern): {unmatched} values")
    return "\n".join(lines)

def distr_analysis_length_stats(dirty_csv, attr_name):
    if attr_name not in dirty_csv.columns:
        return f"Attribute '{attr_name}' not found in dataframe columns: {list(dirty_csv.columns)}"
    
    series = dirty_csv[attr_name]
    values = []
    for v in series:
        if pd.isna(v) or (isinstance(v, str) and v.strip() == ''):
            continue
        values.append(str(v).strip())
    
    if not values:
        return f"Attribute '{attr_name}' has no non-empty string values to analyze length."
    
    lengths = [len(s) for s in values]
    min_len = min(lengths)
    max_len = max(lengths)
    mean_len = float(np.mean(lengths))
    median_len = float(np.median(lengths))
    q1 = float(np.percentile(lengths, 25))
    q3 = float(np.percentile(lengths, 75))
    
    # Samples for min, median, max
    min_sample = next((s for s in values if len(s) == min_len), '')
    sorted_by_len = sorted(values, key=lambda s: len(s))
    median_sample = sorted_by_len[len(sorted_by_len)//2] if sorted_by_len else ''
    max_sample = next((s for s in reversed(sorted_by_len) if len(s) == max_len), '')
    
    lines = [
        f"Attribute: {attr_name}",
        f"Non-empty value count: {len(values)}",
        f"Length min: {min_len} (sample: '{min_sample}')",
        f"Q1 (25th percentile): {q1}",
        f"Median length: {median_len} (sample: '{median_sample}')",
        f"Q3 (75th percentile): {q3}",
        f"Length max: {max_len} (sample: '{max_sample}')",
        f"Mean length: {mean_len:.2f}"
    ]
    return "\n".join(lines)

    numbers = []

            numbers.append(float(found[0]))

    min_num = min(numbers)
    max_num = max(numbers)
    mean_num = float(np.mean(numbers))
    median_num = float(np.median(np.array(numbers)))

        f"Attribute: {attr_name}",

        f"Extracted numbers (first per value):",
        f"  Min: {min_num}, Max: {max_num}, Mean: {mean_num:.2f}, Median: {median_num}",

def distr_analysis_missingness_and_consistency(dirty_csv, attr_name):
    if attr_name not in dirty_csv.columns:
        return f"Attribute '{attr_name}' not found in dataframe columns: {list(dirty_csv.columns)}"
    
    series = dirty_csv[attr_name]
    values = []
    for v in series:
        if pd.isna(v):
            values.append('')
        else:
            values.append(str(v).strip())
    
    total = len(values)
    blank_count = sum(1 for v in values if v == '')
    whitespace_only_count = sum(1 for v in values if v != '' and v.strip() == '')
    leading_trailing_count = sum(1 for v in values if v != '' and v != v.strip())
    
    lines = [
        f"Attribute: {attr_name}",
        f"Total rows: {total}",
        f"Empty/blank values: {blank_count} ({blank_count/total*100:.2f}%)",
        f"Whitespace-only entries: {whitespace_only_count}",
        f"Leading/trailing whitespace occurrences: {leading_trailing_count}"
    ]
    
    # Provide a concrete example if any
    samples = [v for v in values if v != '' and v != v.strip()]
    if samples:
        lines.append(f"Example of a value with whitespace issue: '{samples[0]}'")
    
    return "\n".join(lines)

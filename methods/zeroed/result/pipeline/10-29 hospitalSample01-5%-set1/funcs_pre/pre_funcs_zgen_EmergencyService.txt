def is_clean_yes_no_strict(row, attr):
    # Value of `EmergencyService` is row[attr]
    value = row[attr]
    v = str(value).strip().lower()
    return v in ('yes', 'no')

def is_clean_boolean_like(row, attr):
    # Value of `EmergencyService` is row[attr]
    value = row[attr]
    if isinstance(value, bool):
        return True
    v = str(value).strip().lower()
    return v in ('yes', 'no')

def is_clean_nonempty_emergencyservice(row, attr):
    # Value of `EmergencyService` is row[attr]
    value = row[attr]
    s = str(value)

def is_clean_punctuation_ignored(row, attr):
    # Value of `EmergencyService` is row[attr]
    value = row[attr]
    import string
    s = ''.join(ch for ch in str(value) if ch.isalnum())
